cmake_minimum_required(VERSION 3.24)
project(PureToneCore VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PT_WITH_CODECS "Build FLAC/MP3/SRC playback and analysis" ON)
add_library(puretone SHARED AudioCore/src/DSP.cpp AudioCore/src/Core.cpp)
target_include_directories(puretone PUBLIC AudioCore/include)
set_target_properties(puretone PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  target_compile_options(puretone PRIVATE /W4 /fp:precise)
else()
  target_compile_options(puretone PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
if(PT_WITH_CODECS)
  target_sources(puretone PRIVATE AudioCore/src/Decoder.cpp AudioCore/src/Player.cpp)
  find_path(FLAC_INCLUDE_DIR FLAC/stream_decoder.h REQUIRED)
  find_path(MPG123_INCLUDE_DIR mpg123.h REQUIRED)
  find_path(SOXR_INCLUDE_DIR soxr.h REQUIRED)
  find_library(FLAC_LIBRARY NAMES FLAC REQUIRED)
  find_library(MPG123_LIBRARY NAMES mpg123 REQUIRED)
  find_library(SOXR_LIBRARY NAMES soxr REQUIRED)
  target_include_directories(puretone PRIVATE ${FLAC_INCLUDE_DIR} ${MPG123_INCLUDE_DIR} ${SOXR_INCLUDE_DIR})
  target_link_libraries(puretone PRIVATE ${FLAC_LIBRARY} ${MPG123_LIBRARY} ${SOXR_LIBRARY})
  target_compile_definitions(puretone PRIVATE PT_WITH_CODECS=1)
  if(WIN32)
    target_compile_definitions(puretone PRIVATE FLAC__NO_DLL MPG123_NO_DLL)
    target_link_libraries(puretone PRIVATE shlwapi)
  endif()
endif()
enable_testing()
add_executable(core_tests Tests/CPP/core_tests.cpp)
target_link_libraries(core_tests PRIVATE puretone)
add_test(NAME core_tests COMMAND core_tests)
